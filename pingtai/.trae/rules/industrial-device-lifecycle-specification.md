# ==============================================================================
# Industrial Device Lifecycle Specification（工业设备生命周期规范）
# Version: 1.0.0
# Status: FROZEN
# ==============================================================================

# 核心原则
# ------------------------------------------------------------------------------
# 1. Device 对象创建 != 硬件连接
# 2. Runtime 统一管理设备生命周期
# 3. 禁止在 __init__() 中自动 connect()
# 4. 所有状态转换必须显式执行
# 5. 所有设备必须遵循统一生命周期状态机


# ==============================================================================
# 标准生命周期（严格顺序）
# ==============================================================================

create
    ↓
register
    ↓
allocate
    ↓
connect
    ↓
initialize
    ↓
ready
    ↓
running
    ↓
pause
    ↓
resume
    ↓
stop
    ↓
disconnect
    ↓
release


# ==============================================================================
# 生命周期定义
# ==============================================================================

# create
# ------------------------------------------------------------------------------
# 创建 Device 对象
# 仅创建 Python 对象
# 禁止:
#   - 打开串口
#   - 建立 TCP
#   - 连接 PLC
#   - 启动线程
#   - 占用硬件资源

device = EncoderDevice(...)


# register
# ------------------------------------------------------------------------------
# 注册到 DevicePool / Runtime
# Runtime 开始接管生命周期

device_pool.register(device)


# allocate
# ------------------------------------------------------------------------------
# 分配设备使用权
# ResourceManager 加锁
# 防止多工位抢占设备

resource_manager.allocate("encoder_1")


# connect
# ------------------------------------------------------------------------------
# 建立真实硬件连接
# 允许:
#   - serial.open()
#   - socket.connect()
#   - SDK login()
#
# connect 必须:
#   - 可超时
#   - 可失败
#   - 可重试
#   - 可恢复

device.connect()


# initialize
# ------------------------------------------------------------------------------
# 初始化设备
# 允许:
#   - 参数下发
#   - 回零
#   - 校准
#   - 模式设置
#
# initialize 不允许:
#   - 启动业务动作

device.initialize()


# ready
# ------------------------------------------------------------------------------
# 设备进入 READY 状态
# 表示:
#   - 可被调度
#   - 可执行 Action
#   - Runtime 可分配任务


# running
# ------------------------------------------------------------------------------
# 正在执行任务
# Action / Flow / TestCase 运行中


# pause
# ------------------------------------------------------------------------------
# 暂停运行
# 保持设备状态
# 不释放资源


# resume
# ------------------------------------------------------------------------------
# 从 pause 恢复运行


# stop
# ------------------------------------------------------------------------------
# 停止任务执行
# 停止运动/采集/输出
# 但保持连接


# disconnect
# ------------------------------------------------------------------------------
# 断开硬件连接
# 释放:
#   - serial
#   - socket
#   - SDK session

device.disconnect()


# release
# ------------------------------------------------------------------------------
# 释放资源占用
# ResourceManager 解锁
# DevicePool 回收

resource_manager.release("encoder_1")


# ==============================================================================
# 工业平台强制规范
# ==============================================================================

# 禁止:
# ------------------------------------------------------------------------------
# ❌ __init__() 自动 connect
# ❌ Device 自己管理生命周期
# ❌ connect 内自动 initialize
# ❌ initialize 内自动 running
# ❌ stop 自动 disconnect
# ❌ disconnect 自动 release


# 必须:
# ------------------------------------------------------------------------------
# ✅ Runtime 统一管理生命周期
# ✅ DevicePool 管理设备注册
# ✅ ResourceManager 管理资源分配
# ✅ LifecycleManager 管理启动关闭
# ✅ 所有状态必须可追踪
# ✅ 所有状态变化必须记录日志


# ==============================================================================
# 标准状态枚举
# ==============================================================================

class DeviceState(Enum):
    CREATED = "created"
    REGISTERED = "registered"
    ALLOCATED = "allocated"
    CONNECTING = "connecting"
    CONNECTED = "connected"
    INITIALIZING = "initializing"
    READY = "ready"
    RUNNING = "running"
    PAUSED = "paused"
    STOPPED = "stopped"
    DISCONNECTED = "disconnected"
    RELEASED = "released"
    ERROR = "error"


# ==============================================================================
# Runtime 标准控制方式
# ==============================================================================

runtime.start()
    ↓
device_pool.connect_all()
    ↓
device.initialize()
    ↓
scheduler.dispatch()


# ==============================================================================
# 设计原则
# ==============================================================================

# 组合优于继承
Device → Protocol → Comm

# 生命周期分离
Object Lifecycle != Hardware Lifecycle

# 资源受 Runtime 控制
Runtime → DevicePool → Device

# 状态驱动
所有设备必须状态机化

# 故障可恢复
connect/disconnect/reconnect 必须支持恢复

# 工业可观测性
所有状态变化必须:
#   - log
#   - metrics
#   - trace
#   - event


# ==============================================================================
# 最终原则（工业 Runtime 核心）
# ==============================================================================
工业标准的"三阶段原则"（Construct → Initialize → Start）
# Device 是资源
# Runtime 是控制器
# Lifecycle 是唯一入口
# StateMachine 是唯一状态来源
